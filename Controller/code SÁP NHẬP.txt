import random

def tro_choi_doan_so():
    so_bi_mat = random.randint(1, 100)
    so_lan_doan = 0
    gioi_han = 7
    
    print("=== TRÒ CHƠI ĐOÁN SỐ BÍ MẬT (1 - 100) ===")
    print(f"Bạn có tối đa {gioi_han} lượt đoán. Chúc may mắn!\n")
    
    while so_lan_doan < gioi_han:
        try:
            doan = int(input(f"Lượt {so_lan_doan + 1}/{gioi_han} - Nhập số của bạn: "))
        except ValueError:
            print("❌ Vui lòng nhập một số nguyên hợp lệ!")
            continue
            
        so_lan_doan += 1
        
        if doan < so_bi_mat:
            print("📉 Số bí mật LỚN HƠN số bạn đoán.")
        elif doan > so_bi_mat:
            print("📈 Số bí mật NHỎ HƠN số bạn đoán.")
        else:
            print(f"\n🎉 CHÚC MỪNG! Bạn đã đoán chính xác số {so_bi_mat} sau {so_lan_doan} lượt!")
            break
    else:
        print(f"\n😭 BẠN ĐÃ THUA! Số bí mật đúng là: {so_bi_mat}")

if __name__ == "__main__":
    tro_choi_doan_so()
